pe="peaking";   mid.frequency.value=1000; mid.Q.value=0.9;
  var high = ac.createBiquadFilter(); high.type="highshelf";high.frequency.value=4000;
  var lp   = ac.createBiquadFilter(); lp.type="lowpass";  lp.frequency.value=22050; lp.Q.value=0.7;
  var hp   = ac.createBiquadFilter(); hp.type="highpass"; hp.frequency.value=20;    hp.Q.value=0.7;
  var fader= ac.createGain();
  var an   = ac.createAnalyser(); an.fftSize=1024; an.smoothingTimeConstant=0.5;
  var fxIn = ac.createGain(), fxOut = ac.createGain(), dry = ac.createGain();
  dry.gain.value = 1;
  trim.connect(low); low.connect(mid); mid.connect(high); high.connect(lp); lp.connect(hp);
  hp.connect(fxIn);
  fxIn.connect(dry); dry.connect(fxOut);
  fxOut.connect(fader);
  fader.connect(an); fader.connect(master);
  var fx = buildFx(fxIn, fxOut);
  return { trim:trim, low:low, mid:mid, high:high, lp:lp, hp:hp, fader:fader, an:an,
           fxIn:fxIn, fxOut:fxOut, dry:dry, fx:fx };
}

function makeIR(seconds, decay){
  var sr=ac.sampleRate, n=Math.floor(sr*seconds), b=ac.createBuffer(2,n,sr), c, i, d;
  for(c=0;c<2;c++){
    d=b.getChannelData(c);
    for(i=0;i<n;i++) d[i]=(Math.random()*2-1)*Math.pow(1-i/n, decay);
  }
  return b;
}

function crushCurve(levels){
  var n=2048, c=new Float32Array(n), i, x;
  for(i=0;i<n;i++){ x=i/(n-1)*2-1; c[i]=Math.round(x*levels)/levels; }
  return c;
}

function driveCurve(amount){
  var n=2048, c=new Float32Array(n), i, x, k=amount;
  for(i=0;i<n;i++){ x=i/(n-1)*2-1; c[i]=Math.tanh(x*k)/Math.tanh(k); }
  return c;
}

// LFO reutilizable: osc -> depth -> destino
function lfo(freq, depth, target, type){
  var o=ac.createOscillator(), g=ac.createGain();
  o.type=type||"sine"; o.frequency.value=freq; g.gain.value=depth;
  o.connect(g); g.connect(target); o.start();
  return {osc:o, depth:g};
}

// Construye TODOS los modulos de efecto en paralelo.
// Cada uno termina en su propio gain (0 = apagado) que alimenta fxOut.
function buildFx(input, output){
  var F = {}, mk = function(name, type){ var g=ac.createGain(); g.gain.value=0; g.connect(output); F[name]={out:g, type:type}; return g; };

  // ECHO: delay sincronizado + realimentacion + filtro en el lazo
  var eD=ac.createDelay(4), eF=ac.createGain(), eL=ac.createBiquadFilter();
  eF.gain.value=0.42; eL.type="lowpass"; eL.frequency.value=4200; eD.delayTime.value=0.25;
  input.connect(eD); eD.connect(eL); eL.connect(eF); eF.connect(eD);
  eD.connect(mk("echo","send"));
  F.echo.delay=eD; F.echo.fb=eF;

  // PING PONG: dos delays cruzados en estereo
  var pL=ac.createDelay(4), pR=ac.createDelay(4), panL=ac.createStereoPanner(), panR=ac.createStereoPanner(),
      pF=ac.createGain();
  panL.pan.value=-1; panR.pan.value=1; pF.gain.value=0.45;
  pL.delayTime.value=0.25; pR.delayTime.value=0.25;
  input.connect(pL); pL.connect(panL); panL.connect(mk("pingpong","send"));
  pL.connect(pR); pR.connect(panR); panR.connect(F.pingpong.out);
  pR.connect(pF); pF.connect(pL);
  F.pingpong.dL=pL; F.pingpong.dR=pR; F.pingpong.fb=pF;

  // REVERB: convolucion con impulso generado
  var cv=ac.createConvolver(), pre=ac.createGain();
  cv.buffer=makeIR(2.6, 2.4); pre.gain.value=1;
  input.connect(pre); pre.connect(cv); cv.connect(mk("reverb","send"));

  // SPIRAL: delay corto con mucha realimentacion + pasa altos que sube
  var sD=ac.createDelay(2), sF=ac.createGain(), sH=ac.createBiquadFilter();
  sD.delayTime.value=0.125; sF.gain.value=0.72; sH.type="highpass"; sH.frequency.value=180;
  input.connect(sD); sD.connect(sH); sH.connect(sF); sF.connect(sD);
  sD.connect(mk("spiral","send"));
  F.spiral.delay=sD; F.spiral.fb=sF; F.spiral.hp=sH;

  // FLANGER: delay muy corto modulado + realimentacion
  var fD=ac.createDelay(0.05), fF=ac.createGain();
  fD.delayTime.value=0.005; fF.gain.value=0.55;
  input.connect(fD); fD.connect(fF); fF.connect(fD);
  fD.connect(mk("flanger","insert"));
  F.flanger.lfo=lfo(0.25, 0.0032, fD.delayTime);

  // PHASER: cascada de allpass modulados
  var ap=[], i, prev=input;
  for(i=0;i<4;i++){
    var f=ac.createBiquadFilter(); f.type="allpass"; f.frequency.value=600+i*350; f.Q.value=6;
    prev.connect(f); prev=f; ap.push(f);
  }
  prev.connect(mk("phaser","insert"));
  F.phaser.filters=ap;
  F.phaser.lfo=lfo(0.3, 700, ap[0].frequency);
  for(i=1;i<4;i++) F.phaser.lfo.depth.connect(ap[i].frequency);

  // TREMOLO: ganancia modulada
  var tG=ac.createGain(); tG.gain.value=0.5;
  input.connect(tG); tG.connect(mk("tremolo","insert"));
  F.tremolo.lfo=lfo(4, 0.5, tG.gain);

  // AUTO PAN
  var aP=ac.createStereoPanner();
  input.connect(aP); aP.connect(mk("autopan","insert"));
  F.autopan.lfo=lfo(1, 0.95, aP.pan);

  // WOBBLE: pasa bajos resonante barrido por LFO
  var wF=ac.createBiquadFilter(); wF.type="lowpass"; wF.frequency.value=700; wF.Q.value=12;
  input.connect(wF); wF.connect(mk("wobble","insert"));
  F.wobble.lfo=lfo(4, 620, wF.frequency);
  F.wobble.filt=wF;

  // BIT CRUSH
  var cS=ac.createWaveShaper(); cS.curve=crushCurve(6); cS.oversample="none";
  input.connect(cS); cS.connect(mk("crush","insert"));
  F.crush.shaper=cS;

  // DRIVE
  var dS=ac.createWaveShaper(), dPre=ac.createGain(), dPost=ac.createGain();
  dS.curve=driveCurve(6); dS.oversample="2x"; dPre.gain.value=2.2; dPost.gain.value=0.55;
  input.connect(dPre); dPre.connect(dS); dS.connect(dPost); dPost.connect(mk("drive","insert"));
  F.drive.shaper=dS;

  // RING MOD: multiplicacion por oscilador
  var rG=ac.createGain(); rG.gain.value=0;
  input.connect(rG); rG.connect(mk("ringmod","insert"));
  F.ringmod.lfo=lfo(180, 1, rG.gain);

  return F;
}

function fxBeatSeconds(d){
  var bpm = d.track ? d.track.bpm*d.rate : 128;
  return (60/bpm) * d.fxBeat;
}

function updateFx(id){
  var d=D[id]; if(!d.nodes) return;
  var F=d.nodes.fx, now=ac.currentTime, name=d.fx, k;
  for(k in F) F[k].out.gain.setTargetAtTime(0, now, 0.03);
  if(!name){
    d.nodes.dry.gain.setTargetAtTime(1, now, 0.03);
    $("#fxName"+id).textContent="DRY";
    return;
  }
  var m=F[name], t=fxBeatSeconds(d);
  // parametros dependientes del tempo
  if(name==="echo"){ m.delay.delayTime.setTargetAtTime(t, now, 0.05); m.fb.gain.setTargetAtTime(0.25+d.wet*0.45, now, 0.05); }
  if(name==="pingpong"){ m.dL.delayTime.setTargetAtTime(t, now, 0.05); m.dR.delayTime.setTargetAtTime(t, now, 0.05); m.fb.gain.setTargetAtTime(0.25+d.wet*0.4, now, 0.05); }
  if(name==="spiral"){ m.delay.delayTime.setTargetAtTime(t/2, now, 0.05); m.fb.gain.setTargetAtTime(0.55+d.wet*0.35, now, 0.05); }
  if(name==="flanger"){ m.lfo.osc.frequency.setTargetAtTime(1/(t*4), now, 0.05); }
  if(name==="phaser"){ m.lfo.osc.frequency.setTargetAtTime(1/(t*4), now, 0.05); }
  if(name==="tremolo"){ m.lfo.osc.frequency.setTargetAtTime(1/t, now, 0.05); }
  if(name==="autopan"){ m.lfo.osc.frequency.setTargetAtTime(1/(t*2), now, 0.05); }
  if(name==="wobble"){ m.lfo.osc.frequency.setTargetAtTime(1/t, now, 0.05); }
  if(name==="crush"){ m.shaper.curve=crushCurve(Math.max(2, Math.round(16-d.wet*14))); }
  if(name==="drive"){ m.shaper.curve=driveCurve(1+d.wet*14); }
  if(name==="ringmod"){ m.lfo.osc.frequency.setTargetAtTime(40+d.wet*400, now, 0.05); }

  m.out.gain.setTargetAtTime(m.type==="send" ? d.wet*1.1 : d.wet, now, 0.03);
  d.nodes.dry.gain.setTargetAtTime(m.type==="send" ? 1 : (1-d.wet*0.95), now, 0.03);
  $("#fxName"+id).textContent = name.toUpperCase()+" "+Math.round(d.wet*100)+"%";
}

function setFx(id, name){
  var d=D[id];
  d.fx = (d.fx===name) ? null : name;
  $$('.fx[data-deck="'+id+'"][data-fx]').forEach(function(b){ b.classList.toggle("on", b.dataset.fx===d.fx); });
  updateFx(id);
}

// ROLL: bucle momentaneo que mantiene la pista en tiempo al salir
function toggleRoll(id, frac){
  var d=D[id]; if(!d.track) return;
  var btns=$$('.fx[data-roll][data-deck="'+id+'"]');
  if(d.roll && d.roll.frac===frac){
    var elapsed=(ac.currentTime - d.roll.t0)*d.rate;
    d.loop=null;
    btns.forEach(function(b){ b.classList.remove("on"); });
    var back=d.roll.from + elapsed;
    d.roll=null;
    if(d.playing) startSource(d, clamp(back,0,d.track.buffer.duration-0.05));
    else d.offset=back;
    return;
  }
  var beat=60/d.track.bpm, len=beat*frac, p=pos(d);
  d.loop={start:p, end:p+len, beats:frac};
  d.roll={frac:frac, from:p, t0:ac.currentTime};
  btns.forEach(function(b){ b.classList.toggle("on", Number(b.dataset.roll)===frac); });
  if(d.playing) startSource(d, p);
}

// BRAKE: frenado de vinilo
function brake(id){
  var d=D[id]; if(!d.track||!d.playing||d.braking) return;
  d.braking=true;
  var btn=document.querySelector('.fx[data-brake][data-deck="'+id+'"]');
  btn.classList.add("on");
  var dur=Math.max(0.35, (60/d.track.bpm)*2), t0=performance.now(), r0=d.rate;
  (function tick(){
    var k=clamp((performance.now()-t0)/(dur*1000),0,1);
    var r=r0*(1-k);
    if(d.src) d.src.playbackRate.setTargetAtTime(Math.max(0.02,r), ac.currentTime, 0.02);
    if(k<1) requestAnimationFrame(tick);
    else{
      var p=pos(d);
      stop(d); d.offset=p;
      d.braking=false; btn.classList.remove("on");
      d.rate=d.tempo;
    }
  })();
}

function toMono(buf){
  var n=buf.length, out=new Float32Array(n), ch=buf.numberOfChannels, c, i, d;
  for(c=0;c<ch;c++){ d=buf.getChannelData(c); for(i=0;i<n;i++) out[i]+=d[i]; }
  if(ch>1) for(i=0;i<n;i++) out[i]/=ch;
  return out;
}

var BLK = 256;
function peaks(mono){
  var n = Math.floor(mono.length/BLK), mn=new Float32Array(n), mx=new Float32Array(n), i, j, lo, hi, base, v;
  for(i=0;i<n;i++){
    lo=1; hi=-1; base=i*BLK;
    for(j=0;j<BLK;j++){ v=mono[base+j]; if(v<lo)lo=v; if(v>hi)hi=v; }
    mn[i]=lo; mx[i]=hi;
  }
  return {mn:mn,mx:mx};
}

function onsetEnv(mono, sr){
  var hop=512, n=Math.floor(mono.length/hop), env=new Float32Array(n), prev=0, i, j, s, e, base, v, mx=0;
  for(i=0;i<n;i++){
    s=0; base=i*hop;
    for(j=0;j<hop;j+=2){ v=mono[base+j]; s+=v*v; }
    e=Math.sqrt(s/(hop/2));
    env[i]=Math.max(0,e-prev); prev=e;
    if(env[i]>mx) mx=env[i];
  }
  if(mx>0) for(i=0;i<n;i++) env[i]/=mx;
  return {env:env, fps: sr/hop};
}

function combScore(env,n,lag,phase){
  var s=0,c=0,i,k;
  for(i=phase;i<n;i+=lag){ k=Math.round(i); s+=env[k]||0; c++; }
  return c ? s/Math.sqrt(c) : 0;
}

function bestPhase(env,n,lag){
  var bp=0,bs=-1,p,s;
  for(p=0;p<lag;p+=0.5){ s=combScore(env,n,lag,p); if(s>bs){ bs=s; bp=p; } }
  return {phase:bp, score:bs};
}

function detectBPM(mono, sr){
  var r=onsetEnv(mono,sr), env=r.env, fps=r.fps;
  var n=Math.min(env.length, Math.floor(fps*150));
  var best=126, bestScore=-1, bpm, lag, s, c, k, L, i;
  for(bpm=70;bpm<=190;bpm+=0.5){
    lag=fps*60/bpm; s=0; c=0;
    for(k=1;k<=4;k++){ L=Math.round(lag*k); for(i=L;i<n;i++){ s+=env[i]*env[i-L]; c++; } }
    s = c? s/c : 0;
    s *= (Math.exp(-Math.pow((bpm-125)/45,2))*0.3 + 0.7);
    if(s>bestScore){ bestScore=s; best=bpm; }
  }
  var fb=best, fs=-1, fp=0, rr;
  for(bpm=best-1.2;bpm<=best+1.2;bpm+=0.01){
    lag=fps*60/bpm; rr=bestPhase(env,n,lag);
    if(rr.score>fs){ fs=rr.score; fb=bpm; fp=rr.phase; }
  }
  var cands=[fb/2, fb, fb*2], bo=fb, bos=-1, ci, cb, cl, cr, prior, sc;
  for(ci=0;ci<3;ci++){
    cb=cands[ci];
    if(cb<78||cb>182) continue;
    cl=fps*60/cb; cr=bestPhase(env,n,cl);
    prior=(cb>=95&&cb<=165)?1:0.86;
    sc=cr.score*prior;
    if(sc>bos){ bos=sc; bo=cb; fp=cr.phase; }
  }
  fb=bo;
  var flag=fps*60/fb, pp=fp, ps=-1, p2;
  for(p2=Math.max(0,fp-1); p2<=fp+1; p2+=0.05){
    s=combScore(env,n,flag,p2);
    if(s>ps){ ps=s; pp=p2; }
  }
  return { bpm: Math.round(fb*100)/100, firstBeat: pp/fps };
}

var NOTES=["C","C#","D","D#","E","F","F#","G","G#","A","A#","B"];
var CAMELOT_MAJ=["8B","3B","10B","5B","12B","7B","2B","9B","4B","11B","6B","1B"];
var CAMELOT_MIN=["5A","12A","7A","2A","9A","4A","11A","6A","1A","8A","3A","10A"];
var KS_MAJ=[6.35,2.23,3.48,2.33,4.38,4.09,2.52,5.19,2.39,3.66,2.29,2.88];
var KS_MIN=[6.33,2.68,3.52,5.38,2.60,3.53,2.54,4.75,3.98,2.69,3.34,3.17];

function detectKey(mono, sr){
  var N=4096, hop=16384, chroma=new Float64Array(12);
  var maxS=Math.min(mono.length, sr*100);
  var freqs=[], map=[], oct, p, f, i, b, w=new Float32Array(N);
  for(oct=2; oct<=5; oct++){
    for(p=0;p<12;p++){
      f=440*Math.pow(2,(oct*12+p-57)/12);
      if(f>sr/2.5) continue;
      freqs.push(f); map.push(p);
    }
  }
  for(i=0;i<N;i++) w[i]=0.5-0.5*Math.cos(2*Math.PI*i/(N-1));
  var start, kco, s0, s1, s2, mag;
  for(start=0; start+N<maxS; start+=hop){
    for(b=0;b<freqs.length;b++){
      kco=2*Math.cos(2*Math.PI*freqs[b]/sr);
      s0=0; s1=0; s2=0;
      for(i=0;i<N;i++){ s0=mono[start+i]*w[i]+kco*s1-s2; s2=s1; s1=s0; }
      mag=Math.sqrt(Math.max(0,s1*s1+s2*s2-kco*s1*s2));
      chroma[map[b]]+=mag;
    }
  }
  var sum=0; for(i=0;i<12;i++) sum+=chroma[i];
  if(sum<=0) return {key:"—", camelot:"—", idx:-1, minor:false};
  for(i=0;i<12;i++) chroma[i]/=sum;
  function corr(prof, rot){
    var mx=0,my=0,j,num=0,dx=0,dy=0,a,bb;
    for(j=0;j<12;j++){ mx+=chroma[(j+rot)%12]; my+=prof[j]; }
    mx/=12; my/=12;
    for(j=0;j<12;j++){ a=chroma[(j+rot)%12]-mx; bb=prof[j]-my; num+=a*bb; dx+=a*a; dy+=bb*bb; }
    return num/Math.sqrt((dx*dy)||1e-9);
  }
  var best=-2, idx=0, minor=false, rr, cM, cm;
  for(rr=0;rr<12;rr++){
    cM=corr(KS_MAJ,rr); if(cM>best){best=cM; idx=rr; minor=false;}
    cm=corr(KS_MIN,rr); if(cm>best){best=cm; idx=rr; minor=true;}
  }
  return { key: NOTES[idx]+(minor?"m":""), camelot: minor?CAMELOT_MIN[idx]:CAMELOT_MAJ[idx], idx: idx, mino
