5; ring<0.86; ring+=0.09){
    c.beginPath(); c.arc(cx,cy,R*ring,0,Math.PI*2); c.stroke();
  }

  // marca de rotacion (1 vuelta = 1.8 s, como vinilo)
  var ang = d.track ? (p/1.8)*Math.PI*2 : 0;
  c.save(); c.translate(cx,cy); c.rotate(ang);
  c.fillStyle = d.playing ? main : "#4c5563";
  c.fillRect(-R*0.018, -R*0.86, R*0.036, R*0.52);
  c.beginPath(); c.arc(0,-R*0.86,R*0.05,0,Math.PI*2); c.fill();
  c.restore();

  // centro
  c.beginPath(); c.arc(cx,cy,R*0.33,0,Math.PI*2);
  c.fillStyle="#0a0e14"; c.fill();
  c.strokeStyle="#2a323e"; c.lineWidth=1; c.stroke();

  c.textAlign="center";
  c.fillStyle = d.bend!==0 ? "#f2c14e" : "#cfd7e1";
  c.font="700 "+Math.round(R*0.17)+"px ui-monospace,monospace";
  c.fillText(d.track? fmt(p) : "--:--", cx, cy+R*0.03);
  c.fillStyle="#7d8794";
  c.font=Math.round(R*0.1)+"px ui-monospace,monospace";
  c.fillText(d.bend!==0 ? (d.bend>0?"+BEND":"-BEND") : (d.scrub?"SCRUB":(d.playing?"PLAY":"PAUSE")), cx, cy+R*0.19);
  c.textAlign="left";
}

function drawSpectrum(){
  var cv=$("#spec"), c=fitCanvas(cv); if(!c||!masterAn) return;
  var w=cv.width,h=cv.height;
  var f=new Uint8Array(masterAn.frequencyBinCount);
  masterAn.getByteFrequencyData(f);
  c.fillStyle="#04060a"; c.fillRect(0,0,w,h);
  var bars=56, bw=w/bars, i, idx, v, bh;
  for(i=0;i<bars;i++){
    idx=Math.round(Math.pow(i/(bars-1),1.9)*(f.length-1));
    v=f[idx]/255; bh=v*v*h;
    c.fillStyle = i<26?"#2b7fbf" : (i<40?"#5ddd8f" : (i<50?"#f2c14e":"#ff5b5b"));
    c.fillRect(i*bw, h-bh, Math.max(1,bw-1), bh);
  }
}

function levels(an){
  var buf=new Float32Array(an.fftSize);
  an.getFloatTimeDomainData(buf);
  var s=0,pk=0,i,v,a;
  for(i=0;i<buf.length;i++){ v=buf[i]; s+=v*v; a=Math.abs(v); if(a>pk)pk=a; }
  return { rms:Math.sqrt(s/buf.length), peak:pk };
}

function meterTo(idBar,idVal,v){
  var pct = clamp((20*Math.log10(v+1e-7)+60)/60*100, 0, 100);
  $(idBar).style.width = pct+"%";
  $(idVal).textContent = db(v)+" dB";
}

$$("canvas.jog").forEach(function(cv){
  var d=null, dragging=false, outer=false, lastAng=0, lastT=0, accum=0;
  function angleOf(e){
    var r=cv.getBoundingClientRect();
    return Math.atan2(e.clientY-(r.top+r.height/2), e.clientX-(r.left+r.width/2));
  }
  function radiusOf(e){
    var r=cv.getBoundingClientRect();
    var dx=e.clientX-(r.left+r.width/2), dy=e.clientY-(r.top+r.height/2);
    return Math.sqrt(dx*dx+dy*dy)/(r.width/2);
  }
  cv.addEventListener("pointerdown", function(e){
    d=D[cv.dataset.deck]; if(!d.track) return;
    e.preventDefault();
    dragging=true; outer = radiusOf(e) > 0.62;
    lastAng=angleOf(e); lastT=performance.now(); accum=0;
    if(!outer){ d.scrub=true; }
    if(cv.setPointerCapture) cv.setPointerCapture(e.pointerId);
  });
  cv.addEventListener("pointermove", function(e){
    if(!dragging||!d) return;
    var a=angleOf(e), da=a-lastAng;
    if(da>Math.PI) da-=Math.PI*2;
    if(da<-Math.PI) da+=Math.PI*2;
    lastAng=a;
    var now=performance.now(), dt=Math.max(8, now-lastT); lastT=now;
    if(outer){
      // pitch bend proporcional a la velocidad angular
      var speed = (da/(Math.PI*2)) * (1000/dt);
      var target = clamp(speed*0.11, -0.18, 0.18);
      setBend(d, d.bend*0.55 + target*0.45);
    } else {
      // scrub: 1 vuelta = 1.8 s de audio
      accum += (da/(Math.PI*2))*1.8;
      if(Math.abs(accum) > 0.012){ seek(d.id, pos(d)+accum); accum=0; }
    }
  });
  function release(){
    if(!dragging||!d) return;
    dragging=false;
    if(outer){ setBend(d, 0); }
    d.scrub=false;
    d=null;
  }
  cv.addEventListener("pointerup", release);
  cv.addEventListener("pointercancel", release);
  cv.addEventListener("pointerleave", function(){ if(dragging) release(); });
  cv.addEventListener("dblclick", function(){
    var dd=D[cv.dataset.deck]; if(dd.track) play(cv.dataset.deck);
  });
});
